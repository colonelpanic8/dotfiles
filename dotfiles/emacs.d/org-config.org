#+begin_src emacs-lisp :tangle org-config.el
;;; -*- lexical-binding: t; -*-
(defun imalison:include-file-as-forms (filename)
  "Include the contents of FILENAME into the source as s-expressions."
  (with-temp-buffer
    (insert-file-contents-literally filename)
    (goto-char (point-min))
    (let (forms form)
      (condition-case nil
          (while t
            (setq form (read (current-buffer)))
            (push form forms))
        (end-of-file nil))
      `,@(nreverse forms))))

** Reschedule Past Items to Today
This function finds all incomplete items in the agenda files that are scheduled
in the past and reschedules them to today. Useful for catching up on tasks that
have slipped past their original scheduled date.
#+begin_src emacs-lisp :tangle org-config-config.el
(defun imalison:reschedule-past-to-today ()
  "Reschedule all incomplete items scheduled in the past to today.
Iterates through all org-agenda files and reschedules any TODO items
that have a SCHEDULED date before today to the current date."
  (interactive)
  (let* ((today-time (org-time-string-to-time (format-time-string "<%Y-%m-%d>")))
         (rescheduled-count 0))
    (org-map-entries
     (lambda ()
       (let* ((scheduled (org-entry-get nil "SCHEDULED"))
              (scheduled-time (when scheduled (org-time-string-to-time scheduled))))
         (when (and scheduled-time
                    (not (org-entry-is-done-p))
                    (time-less-p scheduled-time today-time))
           (org-schedule nil ".")
           (cl-incf rescheduled-count))))
     nil
     'agenda)
    (message "Rescheduled %d items to today" rescheduled-count)))

(defun imalison:relevant-to (person)
  (let ((assignee (org-entry-get nil "ASSIGNEE"))
        (completer (org-entry-get nil "COMPLETER")))
    (or
     (null person)
     (string-equal assignee person)
     (string-equal completer person)
     (and
      (null (or assignee completer))
      (or
       (imalison:shared-org-file-p)
       (equal person imalison:org-whoami))))))

(defmacro imalison:assigned-to-me ()
  `(let ((assignee (org-entry-get nil "ASSIGNEE")))
     (or (string-equal assignee imalison:org-whoami)
         (null assignee))))

  (imalison:def-agenda-pred
   (imalison:assigned-to-me)))

(defun org-get-priority-at-point ()
  (save-excursion
    (beginning-of-line)
    (org-back-to-heading t)
    (when (looking-at org-priority-regexp)
	  (let ((ms (match-string 2)))
	    (org-priority-to-value ms)))))

(defmacro imalison:def-agenda-priority-pred (priority)
  `(imalison:def-agenda-pred
    (>= (org-get-priority-at-point) ,priority)))

(cl-defun imalison:org-time-condition-met-p (&key (property "CREATED") (days 30) (future nil))
  (let* ((property-value (org-entry-get (point) property))
         (comparison-time
          (if future
              (time-add (current-time) (days-to-time days))
            (time-subtract (current-time) (days-to-time days))))
         (formatted-time-string (format-time-string "<%Y-%m-%d %H:%M>" comparison-time))
         (compare-time (org-time-string-to-time formatted-time-string))
         (node-time (when property-value (org-time-string-to-time property-value))))
    (when node-time
      (if future
          (time-less-p node-time compare-time)
        (time-less-p compare-time node-time)))))

(defun org-cmp-creation-times (a b)
  (let ((a-created (get-date-created-from-agenda-entry a))
        (b-created (get-date-created-from-agenda-entry b)))
    (imalison:compare-int-list a-created b-created)))

** Use my own default naming scheme for org-headings
First we define a function that will generate a sanitized version of the heading
as its link target.
#+BEGIN_SRC emacs-lisp
(defun imalison:org-get-raw-value (item)
  (when (listp item)
    (let* ((property-list (cadr item)))
      (when property-list (plist-get property-list :raw-value)))))

(defun imalison:sanitize-name (name)
  (replace-regexp-in-string "[^[:alpha:]]" "" (s-downcase name)))

(defun imalison:generate-name (datum cache)
  (let ((raw-value (imalison:org-get-raw-value datum)))
    (if raw-value
        (imalison:sanitize-name raw-value)
      ;; This is the default implementation from org
      (let ((type (org-element-type datum)))
        (format "org%s%d"
                (if type
                    (replace-regexp-in-string "-" "" (symbol-name type))
                    "secondarystring")
                (cl-incf (gethash type cache 0)))))))
#+END_SRC

This function replaces the default naming scheme with a call to
~imalison:generate-name~, and uses a slightly different uniquify approach.
#+BEGIN_SRC emacs-lisp
(use-package ox
  :defer t
  :ensure nil
  :config
  (defun org-export-get-reference (datum info)
    "Return a unique reference for DATUM, as a string.
DATUM is either an element or an object.  INFO is the current
export state, as a plist.  Returned reference consists of
alphanumeric characters only."
    (let ((type (org-element-type datum))
          (cache (or (plist-get info :internal-references)
                     (let ((h (make-hash-table :test #'eq)))
                       (plist-put info :internal-references h)
                       h)))
          (reverse-cache (or (plist-get info :taken-internal-references)
                             (let ((h (make-hash-table :test 'equal)))
                               (plist-put info :taken-internal-references h)
                               h))))
      (or (gethash datum cache)
          (let* ((name (imalison:generate-name datum cache))
                 (number (+ 1 (gethash name reverse-cache -1)))
                 (new-name (format "%s%s" name (if (< 0 number) number ""))))
            (puthash name number reverse-cache)
            (puthash datum new-name cache)
            new-name)))))

** org-node
Replaces org-roam as the notetaking engine.  Same on-disk format --- notes are
just Org files with =id:= links --- but the index lives in hash tables built by
[[https://github.com/meedstrom/org-mem][org-mem]] instead of a SQLite database,
so there is nothing to sync, corrupt, or migrate.  A cold scan of all of
=~/org= takes well under a second.

Unlike org-roam there is no =org-roam-directory= equivalent: org-mem watches
=imalison:org-dir= wholesale, so an ID under =gtd.org= is as reachable as one
under =roam/=, and links between the two produce real backlinks.  If the
completion list ever feels noisy, narrow it with =org-node-filter-fn= rather
than by restricting =org-mem-watch-dirs= (the watch dirs also drive backlinks).
#+begin_src emacs-lisp
(use-package org-mem
  :after org
  :defer 1
  :custom
  ((org-mem-watch-dirs (list (expand-file-name imalison:org-dir)))
   ;; The explicit watch directory is the complete note corpus.  Do not also
   ;; infer directories from `recentf-list' (or agenda/ID state): stale Org
   ;; paths there can otherwise make the initial scan fail after this
   ;; dotfiles checkout moves.
   (org-mem-do-look-everywhere nil)
   (org-mem-do-sync-with-org-id t)
   ;; Drop upstream's ".org_archive" from the default.  The archive files hold
   ;; ~600 dead nodes --- mostly old org-gcal event instances, where a
   ;; recurring "Dentist" or "Flight to San Francisco" contributes one ID per
   ;; occurrence --- so indexing them buys nothing but title collisions and
   ;; completion noise.  Nothing outside the archives links to an ID inside
   ;; them, so no backlink is lost by this.
   (org-mem-suffixes '(".org")))
  :config
  (org-mem-updater-mode +1))

(use-package org-node
  :after org-mem
  ;; Not purely deferred: `org-node-cache-mode' is what keeps the tables warm
  ;; and hooks file saves, so it has to be on before the first lookup.
  :defer 1
  :preface
  (defun imalison:org-node-browse-backlink ()
    "Visit one of the entries whose links point at the ID node at point."
    (interactive)
    (let* ((node (or (org-node-at-point)
                     (user-error "No ID node at point")))
           (entry-alist
            (cl-loop for link in (org-mem-id-links-to-entry node)
                     for source = (org-mem-link-entry link)
                     when source
                     collect (cons (string-join
                                    (org-mem-entry-olpath-with-file-title-or-basename
                                     source)
                                    " > ")
                                   source)))
           (selected-name (completing-read "Select a backlink to visit: "
                                           entry-alist nil t))
           (selected (alist-get selected-name entry-alist nil nil 'string-equal)))
      (when selected (org-node-goto selected))))
  :custom
  (;; Keep creating roam-style "20230813210721-emacs.org" filenames, in the
   ;; directory the existing notes are in.
   (org-node-file-directory-ask (expand-file-name "roam/" imalison:org-dir))
   (org-node-file-timestamp-format "%Y%m%d%H%M%S-")
   (org-node-file-slug-fn #'org-node-slugify-like-roam-default))
  :config
  (progn
    (org-node-cache-mode +1)

    ;; Two overrides on org-node's own prefix map, so that the keys reached via
    ;; either prefix below agree.  `org-node-org-prefix-map' inherits from the
    ;; global one, so setting them here covers both.  Everything else in the
    ;; map keeps upstream's layout -- notably "f" is already `org-node-find'
    ;; and "i" is already `org-node-insert-link', which is why those two no
    ;; longer need binding by hand.
    (keymap-set org-node-global-prefix-map
                ;; Upstream puts `org-node-context-dwim' on "b"; keep "b" as
                ;; the completing-read browser this config has always had.

*** org-roam compatibility
org-roam is kept installed, but purely as a passive library so that the
=*org-roam*= buffer and org-roam-ui keep working.  Its indexer is off:
=org-roam-db-autosync-mode= is never enabled, and org-mem writes the
=org-roam.db= file itself from its own in-memory data.  Nothing here calls
=org-roam-db-sync=.

Deliberately not force-loaded.  Note that bare =:after= would *not* be enough:
use-package loads an =:after= package as soon as its dependency loads, so the
explicit =:defer t= is what keeps org-roam, emacsql and the on-disk database out
of startup until something actually asks for them --- =M-x org-roam-ui-open= or
=M-x org-roam-buffer-toggle=, both autoloaded.  Day-to-day browsing goes through
org-node and never touches SQLite at all.
#+begin_src emacs-lisp
(use-package org-roam
  :if (file-directory-p (expand-file-name "roam/" imalison:org-dir))
  :after org-node
  :defer t
  :custom
  ((org-roam-directory (expand-file-name "roam/" imalison:org-dir))
   ;; org-mem owns the database now.
   (org-roam-db-update-on-save nil))
  :config
  (progn
    (when (version<= "29.0" emacs-version)
      (setq org-roam-database-connector 'sqlite-builtin))
    ;; Serve `org-roam-backlinks-get' etc. out of org-mem's hash tables.
    (org-node-roam-accelerator-mode +1)
    ;; Keep a real org-roam.db on disk for packages that query it directly
    ;; (org-roam-ui).  Only one Emacs session may hold it open.
    (setq org-mem-roamy-do-overwrite-real-db t)
    (org-mem-roamy-db-mode +1)))

** org-fc
#+begin_src emacs-lisp
(use-package org-fc
  :after org
  :bind (("C-c 9" . org-fc-hydra/body)
         ("C-c v c" . imalison:org-capture-vocabulary)
         ("C-c v r" . imalison:org-fc-review-vocabulary)
         ("C-c v f" . imalison:org-vocabulary-fill-pending))
  :preface
  (defvar imalison:org-vocabulary-backends
    '((claude "claude" "-p" "--model" "haiku")
      (codex "codex" "exec" "--ephemeral" "--skip-git-repo-check"
             "--sandbox" "read-only"))
    "Agent CLIs that can look a word up, as (NAME . COMMAND).
The prompt is appended to COMMAND as a single argument, and the JSON
object the agent prints on stdout is used to fill in the card.")

  (defvar imalison:org-vocabulary-backend 'claude
    "Backend in `imalison:org-vocabulary-backends' used to look words up.")

  (defvar imalison:org-vocabulary--captured-id nil
    "ID of the vocabulary card captured by the in-flight capture.")

  (defun imalison:org-capture-vocabulary ()
    "Capture a productive-vocabulary card."
    (interactive)
    (org-capture nil "v"))

  (defun imalison:org-fc-initialize-vocabulary-card ()
    "Initialize the captured entry as an FSRS vocabulary card."
    (require 'org-fc-algo-fsrs)
    (require 'org-fc-type-vocab)
    (let ((org-fc-default-algo 'fsrs6))
      (org-fc-type-vocab-init))
    (save-excursion
      (goto-char (point-min))
      (org-entry-put (point) "VOCAB_FILL" "pending")
      (setq imalison:org-vocabulary--captured-id (org-id-get-create))))

  (defun imalison:org-vocabulary--find (id)
    "Return a marker at the vocabulary card with ID, or nil."
    (with-current-buffer (find-file-noselect imalison:org-vocabulary-file)
      (org-with-wide-buffer
       (let ((position (org-find-property "ID" id)))
         (when position (copy-marker position))))))

  (defun imalison:org-vocabulary-fill-captured ()
    "Fill in the definition of the card the last capture created."
    (let ((id imalison:org-vocabulary--captured-id))
      (setq imalison:org-vocabulary--captured-id nil)
      (when (and id (imalison:org-vocabulary--find id))
        (imalison:org-vocabulary-fill-id id))))

  (defun imalison:org-vocabulary--prompt (word context)
    "Return the prompt asking an agent to define WORD, seen in CONTEXT."
    (concat
     (format "Fill in a productive-vocabulary flashcard for the English word: %s\n\n"
             word)
     "Reply with a single JSON object and nothing else, with exactly these keys:\n"
     "  \"definition\": one concise sentence defining the word, written as a"
     " dictionary gloss. Do not restate the word itself.\n"
     "  \"closest\": the single most ordinary everyday word that comes"
     " closest to the same meaning, as one lowercase word with no punctuation."
     " Pick the plain word someone would otherwise reach for; it should be"
     " close enough to be worth replacing, not an exact synonym.\n"
     "  \"example\": one natural sentence that uses the word correctly and"
     " makes its meaning clear from context.\n\n"
     "Write the definition and example as complete sentences, capitalized and"
     " ending with a period. If the word has several distinct senses, use the"
     " most common one."
     (when context
       (concat "\n\nThe word was encountered here; prefer the sense it has"
               " in this context:\n" context))))

  (defun imalison:org-vocabulary--parse (output)
    "Return the JSON object OUTPUT contains, ignoring prose around it."
    (let* ((start (string-search "{" output))
           (end (cl-position ?} output :from-end t)))
      (when (and start end (< start end))
        (ignore-errors
          (json-parse-string (substring output start (1+ end))
                             :object-type 'alist)))))

  (defun imalison:org-vocabulary-fill-id (id)
    "Ask an agent to define the word of the vocabulary card ID."
    (let* ((marker (or (imalison:org-vocabulary--find id)
                       (error "No vocabulary card with id %s" id)))
           (word (org-with-point-at marker (org-get-heading t t t t)))
           (context (org-with-point-at marker
                      (org-entry-get (point) "VOCAB_CONTEXT")))
           (command (or (alist-get imalison:org-vocabulary-backend
                                   imalison:org-vocabulary-backends)
                        (error "No such vocabulary backend: %s"
                               imalison:org-vocabulary-backend)))
           ;; Run outside any project so the agent picks up no repository
           ;; context, and keep stderr out of the JSON we parse.
           (default-directory temporary-file-directory)
           (errors (generate-new-buffer (format " *vocab %s stderr*" word)))
           (process
            (make-process
             :name (format "vocab-define %s" word)
             :buffer (generate-new-buffer (format " *vocab %s*" word))
             :stderr errors
             :noquery t
             :connection-type 'pipe
             :command (append command
                              (list (imalison:org-vocabulary--prompt
                                     word context)))
             :sentinel
             (lambda (process _event)
               (unless (process-live-p process)
                 (let ((output (with-current-buffer (process-buffer process)
                                 (buffer-string)))
                       (failure (with-current-buffer errors (buffer-string)))
                       (succeeded (eq 0 (process-exit-status process))))
                   (kill-buffer (process-buffer process))
                   (kill-buffer errors)
                   (imalison:org-vocabulary--fill-in
                    id word (and succeeded output) failure)))))))
      ;; The agent CLIs read a prompt from stdin when one is open.
      (process-send-eof process)
      (message "Looking up %s..." word)))

  (defun imalison:org-vocabulary--fill-in (id word output failure)
    "Fill the card ID for WORD in from OUTPUT, or report FAILURE."
    (let ((parsed (and output (imalison:org-vocabulary--parse output)))
          (marker (imalison:org-vocabulary--find id)))
      (cond
       ((not (and (stringp (alist-get 'definition parsed))
                  (stringp (alist-get 'closest parsed))
                  (stringp (alist-get 'example parsed))))
        (imalison:org-vocabulary--set-status id "failed")
        (message "Could not look up %s: %s" word
                 (string-trim (or output failure ""))))
       (marker
        (org-with-point-at marker
          (org-entry-delete (point) "VOCAB_FILL")
          (org-end-of-meta-data t)
          (insert (format "%s\n\nClosest: %s\n\nExample: %s\n\n"
                          (alist-get 'definition parsed)
                          (alist-get 'closest parsed)
                          (alist-get 'example parsed)))
          (save-buffer))
        (message "Filled in the vocabulary card for %s" word)))))

  (defun imalison:org-vocabulary--set-status (id status)
    "Set the VOCAB_FILL property of the card ID to STATUS."
    (let ((marker (imalison:org-vocabulary--find id)))
      (when marker
        (org-with-point-at marker
          (org-entry-put (point) "VOCAB_FILL" status)
          (save-buffer)))))

  (defun imalison:org-vocabulary-fill-pending ()
    "Look up every vocabulary card that has no definition yet."
    (interactive)
    (let ((ids (with-current-buffer
                   (find-file-noselect imalison:org-vocabulary-file)
                 (org-with-wide-buffer
                  (delq nil
                        (org-map-entries
                         (lambda ()
                           (when (org-entry-get (point) "VOCAB_FILL")
                             (org-id-get-create)))))))))
      (if (null ids)
          (message "No vocabulary cards are waiting to be filled in")
        (mapc #'imalison:org-vocabulary-fill-id ids))))

  (defun imalison:org-fc-review-vocabulary ()
    "Review due cards in `imalison:org-vocabulary-file'."
    (interactive)
    (require 'org-fc-algo-fsrs)
    (require 'org-fc-type-vocab)
    (org-fc-review
     `(:paths ,imalison:org-vocabulary-file :order shuffled)))
  :custom
  ((org-fc-default-algo 'fsrs6)
   (org-fc-algo-fsrs6-desired-retention 0.9)
   (org-fc-directories (list imalison:org-vocabulary-file)))
  :config
  (progn
    (require 'org-fc-hydra)
    (require 'org-fc-algo-fsrs)
    (require 'org-fc-type-vocab)
    (add-to-list
     'org-fc-custom-contexts
     `(vocabulary . (:paths ,imalison:org-vocabulary-file
                     :order shuffled))))
  :ensure (:host github :repo "l3kn/org-fc"
           :files ("*.el" "awk" "python" "demo.org")
           :pre-build ("git submodule update --init --recursive")))
